cmake_minimum_required(VERSION 3.26)
project(WaniEngineRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Multimedia DBus)
set(runtime_headers "${WANIENGINE_PLATFORM_ROOT}/qt/runtime")
add_library(waniengine_runtime MODULE "${runtime_headers}/runtime.cpp" "${runtime_headers}/runtime.h" "${runtime_headers}/plugin.cpp")
target_include_directories(waniengine_runtime PRIVATE "${runtime_headers}")
target_link_libraries(waniengine_runtime PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Multimedia Qt6::DBus)
set_target_properties(waniengine_runtime PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml/WaniEngine/Runtime")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qml/WaniEngine/Runtime/qmldir" "module WaniEngine.Runtime\nplugin waniengine_runtime\n")
add_custom_target(wallpaper_package ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${WANIENGINE_PLATFORM_ROOT}/wallpaper" "${CMAKE_CURRENT_BINARY_DIR}/wallpaper" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_BINARY_DIR}/qml/WaniEngine/Runtime" "${CMAKE_CURRENT_BINARY_DIR}/wallpaper/contents/ui/runtime" COMMAND ${CMAKE_COMMAND} -P "${runtime_headers}/package_module.cmake" "${CMAKE_CURRENT_BINARY_DIR}/wallpaper/contents/ui/runtime" DEPENDS waniengine_runtime)
